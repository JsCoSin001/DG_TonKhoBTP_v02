sql  : CREATE TABLE "LichSuCat"
       
       (
       
           "id" INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT,
       
       
       
           "KeHoachCatChiTiet_ID" INTEGER NOT NULL UNIQUE,
       
       
       
           -- Luôn trỏ tới cuộn lẻ thực tế.
       
           -- Với CUON_CHAN, đây là TonCuonLe vừa được tạo
       
           -- trong cùng transaction.
       
           "TonCuonLe_ID" INTEGER NOT NULL,
       
       
       
           "LoaiNguon" TEXT NOT NULL
       
               CHECK ("LoaiNguon" IN ('CUON_CHAN', 'CUON_LE')),
       
       
       
           -- Theo nghiệp vụ hiện tại:
       
           -- chiều dài thực tế phải bằng chiều dài kế hoạch.
       
           "ChieuDaiCat" INTEGER NOT NULL
       
               CHECK ("ChieuDaiCat" > 0),
       
       
       
           -- Trạng thái cuộn trước khi cắt
       
           "SoDauTruoc" INTEGER NOT NULL,
       
           "SoCuoiTruoc" INTEGER NOT NULL,
       
       
       
           -- Mốc của chính đoạn đã cắt
       
           "SoDauCat" INTEGER NOT NULL,
       
           "SoCuoiCat" INTEGER NOT NULL,
       
       
       
           -- Trạng thái phần cuộn còn lại sau khi cắt
       
           "SoDauSau" INTEGER NOT NULL,
       
           "SoCuoiSau" INTEGER NOT NULL,
       
       
       
           -- +1 hoặc -1
       
           "HeSoChieu" INTEGER NOT NULL
       
               CHECK ("HeSoChieu" IN (-1, 1)),
       
       
       
           -- Nội dung tem/mã thực tế người lao động quét tại B3.
       
           -- Dùng để audit, không thay cho khóa TonCuonLe_ID.
       
           "MaQuetThucTe" TEXT NOT NULL,
       
       
       
           "NguoiThucHien" TEXT NOT NULL,
       
       
       
           "ThoiGianThucHien" TEXT NOT NULL
       
               DEFAULT (strftime('%Y-%m-%d %H:%M:%S', 'now', 'localtime')),
       
       
       
           "GhiChu" TEXT, "DieuChinhThongSo" TEXT DEFAULT NULL,
       
       
       
           FOREIGN KEY ("KeHoachCatChiTiet_ID")
       
               REFERENCES "KeHoachCatChiTiet"("id")
       
               ON UPDATE CASCADE
       
               ON DELETE CASCADE,
       
       
       
           FOREIGN KEY ("TonCuonLe_ID")
       
               REFERENCES "TonCuonLe"("id")
       
               ON UPDATE CASCADE
       
               ON DELETE CASCADE
       
       )

name : TonCuonLe
sql  : CREATE TABLE "TonCuonLe"
       
       (
       
           "id" INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT,
       
       
       
           -- Mã nhận diện duy nhất của cuộn vật lý khi đã trở thành cuộn lẻ.
       
           "MaCuon" TEXT NOT NULL UNIQUE,
       
       
       
           -- TTThanhPham tương ứng với MaBin/nghiệp vụ của cuộn.
       
           -- Không UNIQUE vì có thể nhiều cuộn vật lý cùng liên quan
       
           -- tới một TTThanhPham trong trường hợp cuộn chẵn.
       
           "TTThanhPham_ID" INTEGER NOT NULL,
       
       
       
           -- Dòng TTCuonDay nguồn.
       
           -- Với cuộn lẻ nhập kho: TTCuonDay đã tạo ra cuộn lẻ này.
       
           -- Với cuộn chẵn -> lẻ: TTCuonDay chứa tồn cuộn chẵn nguồn.
       
           "TTCuonDay_ID" INTEGER NOT NULL,
       
       
       
           -- Trạng thái hiện tại của cuộn lẻ.
       
           "SoDau" INTEGER NOT NULL,
       
           "SoCuoi" INTEGER NOT NULL,
       
       
       
           -- ACTIVE: còn chiều dài
       
           -- HET:    đã sử dụng hết
       
           "TrangThai" TEXT NOT NULL DEFAULT 'ACTIVE'
       
               CHECK ("TrangThai" IN ('ACTIVE', 'HET')),
       
       
       
           "GhiChu" TEXT,
       
       
       
           "DateInsert" TEXT NOT NULL
       
               DEFAULT (strftime('%Y-%m-%d %H:%M:%S', 'now', 'localtime')),
       
       
       
           FOREIGN KEY ("TTThanhPham_ID")
       
               REFERENCES "TTThanhPham"("id")
       
               ON UPDATE CASCADE
       
               ON DELETE CASCADE,
       
       
       
           FOREIGN KEY ("TTCuonDay_ID")
       
               REFERENCES "TTCuonDay"("id")
       
               ON UPDATE CASCADE
       
               ON DELETE CASCADE
       
       )
